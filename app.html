        // Fonctions utilitaires

        function saveGameState() {

            localStorage.setItem('tribunalIA_gameState', JSON.stringify({

                currentRole,

                playedCards,

                totalScore,

                timestamp: Date.now()

            }));

        }


        function loadGameState() {

            const saved = localStorage.getItem('tribunalIA_gameState');

            if (saved) {

                const state = JSON.parse(saved);

                // Vérifier si la sauvegarde n'est pas trop ancienne (24h)

                if (Date.now() - state.timestamp < 86400000) {

                    return state;

                }

            }

            return null;

        }


        function selectRole(role) {

            currentRole = role;

            currentCards = [...cardSets[role]];

            shuffleCards();

            

            document.getElementById('gameArea').classList.remove('hidden');

            document.getElementById('roleTitle').textContent = getRoleTitle(role);

            document.getElementById('triggerIncident').disabled = false;

            

            displayCards();

            updateCounters();

            saveGameState();

        }


        function triggerIncident() {

            if (!incidentCards || incidentCards.length === 0) {

                showNotification('⚠️ Aucune carte incident disponible !');

                return;

            }

            

            if (currentCards.length === 0) {

                showNotification('⚠️ Aucune carte disponible pour déclencher un incident !');

                return;

            }

            

            if (activeIncident) {

                showNotification('⚠️ Un incident est déjà en cours ! Résolvez-le d\'abord.');

                return;

            }

            

            const randomIncident = incidentCards[Math.floor(Math.random() * incidentCards.length)];

            

            const incidentCard = {

                ...randomIncident,

                isTriggeredIncident: true,

                id: 'incident_' + Date.now()

            };

            

            currentCards.unshift(incidentCard);

            activeIncident = true;

            

            const triggerBtn = document.getElementById('triggerIncident');

            triggerBtn.disabled = true;

            triggerBtn.textContent = '⚡ Incident Actif';

            triggerBtn.classList.add('bg-red-700');

            

            displayCards();

            

            setTimeout(() => {

                const cardContainer = document.querySelector('#card-container-0');

                if (cardContainer) {

                    cardContainer.scrollIntoView({ behavior: 'smooth', block: 'center' });

                    cardContainer.style.animation = 'pulse 2s infinite';

                }

            }, 100);

            

            showNotification('⚡ INCIDENT DÉCLENCHÉ ! Une carte d\'urgence a été ajoutée au jeu.');

        }


        function showNotification(message) {

            const notification = document.createElement('div');

            notification.className = 'fixed top-4 right-4 bg-red-600 text-white px-6 py-3 rounded-lg shadow-lg z-50 text-lg font-semibold';

            notification.textContent = message;

            document.body.appendChild(notification);

            

            setTimeout(() => {

                notification.remove();

            }, 4000);

        }

            return titles[role] || 'Rôle Inconnu';

        function shuffleCards() {

            for (let i = currentCards.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [currentCards[i], currentCards[j]] = [currentCards[j], currentCards[i]];

            }

            if (currentCards.length > 0) {

                displayCards();

            }

        }


        function displayCards() {

            const container = document.getElementById('cardsContainer');

            if (!container) return;

            

            container.innerHTML = '';

            

            if (currentCards.length === 0) {

                container.innerHTML = '<div class="col-span-full text-center text-gray-500 text-xl">Aucune carte disponible</div>';

                return;

            }

            

            currentCards.forEach((card, index) => {

                const cardElement = createCardElement(card, index);

                container.appendChild(cardElement);

            });

        }

                });

            });

            

            return cardDiv;

        }


        function flipCard(index) {

            const card = document.getElementById(`card-${index}`);

            if (card) {

                card.classList.toggle('flipped');

            }

        }


        function selectOption(cardIndex, optionIndex) {

            const card = currentCards[cardIndex];

            if (!card) return;

            

            totalScore += (card.points || 0);

            playedCards++;

            

            const isIncidentResolved = card.type === 'incident' || card.isTriggeredIncident;

            

            const cardElement = document.getElementById(`card-${cardIndex}`);

            if (cardElement) {

                cardElement.style.transform = 'scale(1.1)';

                cardElement.style.opacity = '0.7';

            }

            

            setTimeout(() => {

                currentCards.splice(cardIndex, 1);

                displayCards();

                updateCounters();

                saveGameState();

                

                if (isIncidentResolved) {

                    activeIncident = false;

                    const triggerBtn = document.getElementById('triggerIncident');

                    if (triggerBtn) {

                        triggerBtn.disabled = false;

                        triggerBtn.textContent = '🎯 Déclencher Incident';

                        triggerBtn.classList.remove('bg-red-700');

        function triggerFullscreenIncident(card) {

            currentIncidentCard = card;

            

            const titleEl = document.getElementById('incidentTitle');

            const scenarioEl = document.getElementById('incidentScenario');

            const dilemmaEl = document.getElementById('incidentDilemma');

            const optionsContainer = document.getElementById('incidentOptions');

            

            if (titleEl) titleEl.textContent = card.title;

            if (scenarioEl) scenarioEl.textContent = card.scenario;

            if (dilemmaEl) dilemmaEl.textContent = card.dilemma;

            

            if (optionsContainer) {

                optionsContainer.innerHTML = '';

                

                card.options.forEach((option, index) => {

                    const optionDiv = document.createElement('div');

                    optionDiv.className = 'bg-white/20 rounded-xl p-6 text-2xl font-bold';

                    optionDiv.textContent = `${index + 1}. ${option}`;

                    optionsContainer.appendChild(optionDiv);

                });

                

                if (card.discussion) {

                    const discussionDiv = document.createElement('div');

                    discussionDiv.className = 'bg-yellow-500/20 rounded-2xl p-6 mt-6 border-2 border-yellow-400';

                    discussionDiv.innerHTML = `

                        <h3 class="text-2xl font-bold text-yellow-300 mb-4">${card.discussion.prompt}</h3>

                        <div class="space-y-3 mb-4">

                            ${card.discussion.questions.map(question => `

                                <div class="bg-white/10 rounded-lg p-4">

                                    <p class="text-xl font-semibold">${question}</p>

                                </div>

                            `).join('')}

                        </div>

                        <div class="bg-red-500/30 rounded-lg p-4 border border-red-400">

                            <p class="text-xl font-bold text-red-200">${card.discussion.consequences}</p>

                        </div>

                    `;

                    optionsContainer.appendChild(discussionDiv);

                }

            }

            

            document.getElementById('fullscreenIncident').classList.remove('hidden');

            

            setTimeout(() => {

                const modal = document.getElementById('fullscreenIncident');

                if (modal) {

                    modal.style.animation = 'pulse 0.5s ease-in-out';

                }

            }, 100);

        }

            if (!currentIncidentCard || !currentIncidentCard.discussion) return;

            

            const timerDiv = document.createElement('div');

            timerDiv.id = 'discussionTimer';

            document.body.appendChild(timerDiv);

            instructionsDiv.id = 'discussionInstructions';

            instructionsDiv.className = 'fixed top-20 left-4 right-4 bg-yellow-500/95 text-black p-6 rounded-xl text-lg font-semibold z-50 shadow-lg max-w-md';

            instructionsDiv.innerHTML = `

                <h3 class="text-xl font-bold mb-3">🎯 Discussion sur l'IA Éducative</h3>

                <div class="space-y-2 text-base">

                    <p>• Chaque participant prend un rôle</p>

                    <p>• Débat de 5 minutes chronométrées</p>

                    <p>• Focus sur les enjeux de l'IA à l'école</p>

                    <p>• Trouvez un consensus si possible</p>

                </div>

                <button id="endDiscussionBtn" class="mt-4 bg-red-600 text-white px-4 py-2 rounded-lg hover:bg-red-700 transition-colors">

                    Terminer Discussion

                </button>

            `;

            document.body.appendChild(instructionsDiv);

            

            document.getElementById('endDiscussionBtn').addEventListener('click', endDiscussion);

            

            let timeLeft = 300;

            const timerInterval = setInterval(() => {

                timeLeft--;

                const minutes = Math.floor(timeLeft / 60);

                const seconds = timeLeft % 60;

                timerDiv.innerHTML = `⏱️ Discussion: ${minutes}:${seconds.toString().padStart(2, '0')}`;

                

                if (timeLeft <= 0) {

                    clearInterval(timerInterval);

                    timerDiv.innerHTML = '⏰ TEMPS ÉCOULÉ !';

                    timerDiv.className = 'fixed top-4 left-4 bg-red-600 text-white px-6 py-3 rounded-xl text-xl font-bold z-50 shadow-lg animate-pulse';

                }

            }, 1000);

            

            window.discussionInterval = timerInterval;

        }


        function endDiscussion() {

            const timer = document.getElementById('discussionTimer');

            const instructions = document.getElementById('discussionInstructions');

            if (timer) timer.remove();

            if (instructions) instructions.remove();

                clearInterval(window.discussionInterval);

            }

            

            showNotification('✅ Discussion terminée ! Merci pour vos échanges sur l\'IA éducative.');

        }
